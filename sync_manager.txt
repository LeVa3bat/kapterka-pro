    65	    fun startSyncForUnit(unitKey: String, callsign: String, unitName: String) {
    66	        val cleanKey = unitKey.trim()
    67	        if (cleanKey.isEmpty()) return
    68	
    69	        if (activeUnitKey == cleanKey && listeners.isNotEmpty()) {
    70	            // Already listening for this unit, just send presence ping
    71	            sendPresencePing(cleanKey, callsign, unitName)
    72	            return
    73	        }
    74	
    75	        stopSync()
    76	        activeUnitKey = cleanKey
    77	        _syncState.value = _syncState.value.copy(
    78	            isSyncing = true,
    79	            syncMessage = "Подключение к каналу подразделения [$cleanKey]..."
    80	        )
    81	
    82	        
    83	            val db = firestore
    84	        val unitRef = db.collection("units").document(unitKey)
    85	        
    86	        try {
    87	            registerUnitListeners(unitKey)
    88	        } catch(e: Exception) {}
    89	        
    90	        _syncState.value = _syncState.value.copy(
    91	            isSyncing = false,
    92	            isOnline = true,
    93	            syncMessage = "Подключено к подразделению"
    94	        )
    95	    }

    97	    private fun registerUnitListeners(unitKey: String) {
    98	        val db = firestore
    99	        val unitRef = db.collection("units").document(unitKey)
   100	
   101	        val pointsReg = unitRef.collection("warehouse_points").addSnapshotListener { snap, e ->
   102	            if (e != null || snap == null) return@addSnapshotListener
   103	            scope.launch(Dispatchers.IO) {
   104	                for (dc in snap.documentChanges) {
   105	                    val p = WarehousePoint(
   106	                        id = dc.document.id,
   107	                        name = dc.document.getString("name") ?: "",
   108	                        description = dc.document.getString("description") ?: "",
   109	                        isBase = dc.document.getBoolean("isBase") ?: false,
   110	                        orderIndex = dc.document.getLong("orderIndex")?.toInt() ?: 0,
   111	                        createdAt = dc.document.getLong("createdAt") ?: 0L
   112	                    )
   113	                    if (dc.type == DocumentChange.Type.REMOVED) {
   114	                        dao.deletePoint(p.id)
   115	                    } else {
   116	                        dao.insertPoint(p)
   117	                    }
   118	                }
   119	            }
   120	        }
   121	        listeners.add(pointsReg)
   122	
   123	        val itemsReg = unitRef.collection("inventory_items").addSnapshotListener { snap, e ->
   124	            if (e != null || snap == null) return@addSnapshotListener
   125	            scope.launch(Dispatchers.IO) {
   126	                for (dc in snap.documentChanges) {
   127	                    val item = InventoryItem(
   128	                        id = dc.document.id,
   129	                        name = dc.document.getString("name") ?: "",
   130	                        serviceCategory = dc.document.getString("serviceCategory") ?: "",
   131	                        subType = dc.document.getString("subType") ?: "",
   132	                        unit = dc.document.getString("unit") ?: "шт.",
   133	                        categoryClass = dc.document.getString("categoryClass") ?: "Кат. 1",
   134	                        standardCode = dc.document.getString("standardCode") ?: "",
   135	                        isCustom = dc.document.getBoolean("isCustom") ?: false
   136	                    )
   137	                    if (dc.type == DocumentChange.Type.REMOVED) {
   138	                        dao.deleteItem(item.id)
   139	                    } else {
   140	                        dao.insertItem(item)
   141	                    }
   142	                }
   143	            }
   144	        }
   145	        listeners.add(itemsReg)
   146	
   147	        val stockReg = unitRef.collection("stock_records").addSnapshotListener { snap, e ->
   148	            if (e != null || snap == null) return@addSnapshotListener
   149	            scope.launch(Dispatchers.IO) {
   150	                for (dc in snap.documentChanges) {
   151	                    val s = StockRecord(
   152	                        pointId = dc.document.getString("pointId") ?: "",
   153	                        itemId = dc.document.getString("itemId") ?: "",
   154	                        quantity = dc.document.getLong("quantity")?.toInt() ?: 0,
   155	                        incomeTotal = dc.document.getLong("incomeTotal")?.toInt() ?: 0,
   156	                        expenseTotal = dc.document.getLong("expenseTotal")?.toInt() ?: 0,
   157	                        lastUpdated = dc.document.getLong("lastUpdated") ?: 0L
   158	                    )
   159	                    if (dc.type != DocumentChange.Type.REMOVED) {
   160	                        dao.insertOrUpdateStock(s)
   161	                    }
   162	                }
   163	            }
   164	        }
   165	        listeners.add(stockReg)
   166	
   167	        val opReg = unitRef.collection("operation_records").addSnapshotListener { snap, e ->
   168	            if (e != null || snap == null) return@addSnapshotListener
   169	            scope.launch(Dispatchers.IO) {
   170	                for (dc in snap.documentChanges) {
   171	                    val opTypeStr = dc.document.getString("type") ?: "INCOME"
   172	                    val type = try { OperationType.valueOf(opTypeStr) } catch (ex: Exception) { OperationType.INCOME }
   173	                    val op = OperationRecord(
   174	                        id = dc.document.id,
   175	                        type = type,
   176	                        fromPointName = dc.document.getString("fromPointName") ?: "",
   177	                        toPointName = dc.document.getString("toPointName") ?: "",
   178	                        docNumber = dc.document.getString("docNumber") ?: "",
   179	                        responsiblePerson = dc.document.getString("responsiblePerson") ?: "",
   180	                        comment = dc.document.getString("comment") ?: "",
   181	                        timestamp = dc.document.getLong("timestamp") ?: 0L,
   182	                        itemsSummary = dc.document.getString("itemsSummary") ?: "",
   183	                        itemsJson = dc.document.getString("itemsJson") ?: ""
   184	                    )
   185	                    if (dc.type == DocumentChange.Type.REMOVED) {
   186	                        dao.deleteOperation(op.id)
   187	                    } else {
   188	                        dao.insertOperation(op)
   189	                    }
   190	                }
   191	            }
   192	        }
   193	        listeners.add(opReg)
   194	
   195	        val reqReg = unitRef.collection("requisitions").addSnapshotListener { snap, e ->
   196	            if (e != null || snap == null) return@addSnapshotListener
   197	            scope.launch(Dispatchers.IO) {
   198	                for (dc in snap.documentChanges) {
   199	                    val statusStr = dc.document.getString("status") ?: "PENDING"
   200	                    val status = try { RequestStatus.valueOf(statusStr) } catch (ex: Exception) { RequestStatus.PENDING }
   201	                    val req = RequisitionRequest(
   202	                        id = dc.document.id,
   203	                        pointName = dc.document.getString("pointName") ?: "",
   204	                        applicantName = dc.document.getString("applicantName") ?: "",
   205	                        status = status,
   206	                        comment = dc.document.getString("comment") ?: "",
   207	                        timestamp = dc.document.getLong("timestamp") ?: 0L,
   208	                        itemsSummary = dc.document.getString("itemsSummary") ?: "",
   209	                        itemsJson = dc.document.getString("itemsJson") ?: ""
   210	                    )
   211	                    if (dc.type == DocumentChange.Type.REMOVED) {
   212	                        dao.deleteRequisition(req.id)
   213	                    } else {
   214	                        dao.insertRequisition(req)
   215	                    }
   216	                }
   217	            }
   218	        }
   219	        listeners.add(reqReg)
   220	        
   221	        val presReg = unitRef.collection("devices").addSnapshotListener { snap, e ->
   222	            if (e != null || snap == null) return@addSnapshotListener
   223	            _syncState.value = _syncState.value.copy(connectedDevicesCount = snap.documents.size)
   224	        }
   225	        listeners.add(presReg)
   226	    }

   228	    private fun sendPresencePing(unitKey: String, callsign: String, unitName: String) {
   229	        val db = firestore
   230	        val unitRef = db.collection("units").document(unitKey)
   231	        val data = hashMapOf(
   232	            "deviceId" to deviceId,
   233	            "callsign" to callsign,
   234	            "unitName" to unitName,
   235	            "deviceModel" to (Build.MANUFACTURER + " " + Build.MODEL),
   236	            "timestamp" to FieldValue.serverTimestamp(),
   237	            "timestampMillis" to System.currentTimeMillis()
   238	        )
   239	        unitRef.collection("devices").document(deviceId).set(data, SetOptions.merge())
   240	        val uData = hashMapOf(
   241	            "unitKey" to unitKey,
   242	            "unitName" to unitName,
   243	            "lastActivity" to FieldValue.serverTimestamp()
   244	        )
   245	        unitRef.set(uData, SetOptions.merge())
   246	    }

   248	    suspend fun pushAllLocalData(unitKey: String) {
   249	        if (unitKey.isEmpty()) return
   250	        val db = firestore
   251	        _syncState.value = _syncState.value.copy(isSyncing = true, syncMessage = "Отправка локальных данных в облако...")
   252	        try {
   253	            dao.getAllPoints().first().forEach { p -> pushWarehousePointAsync(unitKey, p) }
   254	            dao.getAllItems().first().forEach { i -> pushInventoryItemAsync(unitKey, i) }
   255	            dao.getAllStockRecords().first().forEach { s -> pushStockRecordAsync(unitKey, s) }
   256	            dao.getAllOperations().first().forEach { o -> pushOperationAsync(unitKey, o, emptyList()) }
   257	            dao.getAllRequisitions().first().forEach { r -> pushRequisitionAsync(unitKey, r) }
   258	
   259	            _syncState.value = _syncState.value.copy(isSyncing = false, syncMessage = "Синхронизация завершена")
   260	        } catch (e: Exception) {
   261	            _syncState.value = _syncState.value.copy(isSyncing = false, syncMessage = "Ошибка синхронизации")
   262	        }
   263	    }

   265	    fun pushOperationAsync(unitKey: String, op: OperationRecord, updatedStocks: List<StockRecord>) {
   266	        if (unitKey.isEmpty()) return
   267	        scope.launch(Dispatchers.IO) {
   268	            
   269	                val db = firestore
   270	                val unitRef = db.collection("units").document(unitKey)
   271	                unitRef.collection("operation_records").document(op.id).set(
   272	                    hashMapOf(
   273	                        "id" to op.id,
   274	                        "type" to op.type.name,
   275	                        "fromPointName" to op.fromPointName,
   276	                        "toPointName" to op.toPointName,
   277	                        "docNumber" to op.docNumber,
   278	                        "responsiblePerson" to op.responsiblePerson,
   279	                        "comment" to op.comment,
   280	                        "timestamp" to op.timestamp,
   281	                        "itemsSummary" to op.itemsSummary,
   282	                        "itemsJson" to op.itemsJson
   283	                    ),
   284	                    SetOptions.merge()
   285	                )
   286	
   287	                for (s in updatedStocks) {
   288	                    val docId = "${s.pointId}___${s.itemId}"
   289	                    unitRef.collection("stock_records").document(docId).set(
   290	                        hashMapOf(
   291	                            "pointId" to s.pointId,
   292	                            "itemId" to s.itemId,
   293	                            "quantity" to s.quantity,
   294	                            "incomeTotal" to s.incomeTotal,
   295	                            "expenseTotal" to s.expenseTotal,
   296	                            "lastUpdated" to s.lastUpdated
   297	                        ),
   298	                        SetOptions.merge()
   299	                    )
   300	                }
   301	
   302	                _syncState.value = _syncState.value.copy(
   303	                    lastSyncTime = System.currentTimeMillis(),
   304	                    isOnline = true,
   305	                    syncMessage = "Операция синхронизирована"
   306	                )
   307	            } catch (e: Exception) {
   308	                Log.w(TAG, "Failed pushing op live, saved locally", e)
   309	            }
   310	        }
   311	    }

   462	    fun stopSync() {
   463	        for (l in listeners) {
   464	            try {
   465	                l.remove()
   466	            } catch (_: Exception) {}
   467	        }
   468	        listeners.clear()
   469	    }

   471	    fun clearCloudDataAsync(unitKey: String) {
   472	        scope.launch(Dispatchers.IO) {
   473	            try {
   474	                val cleanKey = unitKey.trim()
   475	                if (cleanKey.isEmpty()) return@launch
   476	                val db = firestore
   477	                val unitRef = db.collection("units").document(cleanKey)
   478	                val collectionsToClear = listOf("stock_records", "operation_records", "requisitions")
   479	                for (col in collectionsToClear) {
   480	                    val snapshot = unitRef.collection(col).get().await()
   481	                    for (doc in snapshot.documents) {
   482	                        doc.reference.delete().await()
   483	                    }
   484	                }
   485	            } catch (e: Exception) {
   486	                Log.e(TAG, "Error clearing cloud data", e)
   487	            }
   488	        }
   489	    }
